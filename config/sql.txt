	  SELECT (t.a_set_codigo)::integer AS set_id,

CREATE OR REPLACE VIEW public.v_spl_cargos_mesa AS
 SELECT t.matricula,
    t.ini_nome,
    t.crg_nome,
    t.crg_ordem
   FROM dblink('host=prod-postgresql-spl.cac3p2a0zcak.sa-east-1.rds.amazonaws.com dbname=pro user=mscmc password=camara'::text, 'select distinct u.usu_matricula, v.ini_nome, cv.crg_nome, cv.crg_ordem
from spl.composicao c
	inner join spl.cargo_vereador cv ON
		c.crg_id = cv.crg_id
	inner join spl.vereador v ON
		c.ver_id = v.ver_id
	inner join spl.usuario u ON
		v.ver_id = u.ver_id
where con_id in (select con_id
		from spl.conjunto_vereadores
		where ini_ativa=true
			and tin_id in (select tin_id
					from spl.tipo_iniciativa
					where tin_nome ilike ''mesa%''
                                      )
		)
order by cv.crg_ordem'::text) t(matricula integer, ini_nome text, crg_nome character varying(50), crg_ordem integer);

    OWNER TO mscmc;
